RouterOS Configuration

# VLAN Configuration on MikroTik

# Create VLAN interfaces on bridge
/interface vlan
add interface=bridge1 name=vlan10 vlan-id=10
add interface=bridge1 name=vlan20 vlan-id=20
add interface=bridge1 name=vlan30 vlan-id=30
add interface=bridge1 name=vlan40 vlan-id=40
add interface=bridge1 name=vlan50 vlan-id=50
add interface=bridge1 name=vlan60 vlan-id=60
add interface=bridge1 name=vlan70 vlan-id=70
add interface=bridge1 name=vlan80 vlan-id=80

# Assign IP addresses to VLANs
/ip address
add address=192.168.10.1/24 interface=vlan10 network=192.168.10.0
add address=192.168.20.1/24 interface=vlan20 network=192.168.20.0
add address=192.168.30.1/24 interface=vlan30 network=192.168.30.0
add address=192.168.40.1/24 interface=vlan40 network=192.168.40.0
add address=192.168.50.1/24 interface=vlan50 network=192.168.50.0
add address=192.168.60.1/24 interface=vlan60 network=192.168.60.0
add address=192.168.70.1/24 interface=vlan70 network=192.168.70.0
add address=192.168.80.1/24 interface=vlan80 network=192.168.80.0

# Configure trunk port on switch
/interface bridge vlan
add bridge=bridge1 tagged=ether1,ether2 vlan-ids=10,20,30,40,50,60,70,80
# PCC Load Balancing with Failover

# Mark connections from each ISP
/ip firewall mangle
add chain=prerouting in-interface=isp1 action=mark-connection \
    new-connection-mark=isp1_conn
add chain=prerouting in-interface=isp2 action=mark-connection \
    new-connection-mark=isp2_conn

# PCC load balancing for outgoing traffic
add chain=prerouting dst-address-type=!local in-interface=lan \
    action=mark-connection new-connection-mark=isp1_conn \
    per-connection-classifier=both-addresses:2/0
add chain=prerouting dst-address-type=!local in-interface=lan \
    action=mark-connection new-connection-mark=isp2_conn \
    per-connection-classifier=both-addresses:2/1

# Mark routing for marked connections
add chain=prerouting connection-mark=isp1_conn in-interface=lan \
    action=mark-routing new-routing-mark=to_isp1
add chain=prerouting connection-mark=isp2_conn in-interface=lan \
    action=mark-routing new-routing-mark=to_isp2

# Add gateway routes with distance for failover
/ip route
add gateway=1.1.1.1 routing-mark=to_isp1 distance=1
add gateway=2.2.2.2 routing-mark=to_isp2 distance=1
add gateway=1.1.1.1 distance=1
add gateway=2.2.2.2 distance=2

# Netwatch for failover detection
/tool netwatch
add host=8.8.8.8 interval=30s down-script="/ip route set [find gateway=1.1.1.1] disabled=yes" \
    up-script="/ip route set [find gateway=1.1.1.1] disabled=no"
# Firewall Security Rules

# Drop invalid connections
/ip firewall filter
add chain=input connection-state=invalid action=drop
add chain=forward connection-state=invalid action=drop

# Allow established/related connections
add chain=input connection-state=established,related action=accept
add chain=forward connection-state=established,related action=accept

# Allow ICMP (ping)
add chain=input protocol=icmp action=accept

# Allow specific services from management VLAN only
add chain=input protocol=tcp dst-port=22 src-address=192.168.10.0/24 action=accept
add chain=input protocol=tcp dst-port=8291 src-address=192.168.10.0/24 action=accept

# Inter-VLAN firewall rules
add chain=forward src-address=192.168.30.0/24 dst-address=192.168.10.0/24 \
    action=drop log=yes comment="Block guest to management"
add chain=forward src-address=192.168.60.0/24 dst-address=192.168.40.0/24 \
    action=drop comment="Block IoT to servers"

# Allow specific inter-VLAN communication
add chain=forward src-address=192.168.20.0/24 dst-address=192.168.40.0/24 \
    protocol=tcp dst-port=80,443 action=accept comment="Employees to web servers"

# Drop everything else
add chain=input action=drop
add chain=forward action=drop
# DHCP Server Setup per VLAN

# Create DHCP pools
/ip pool
add name=dhcp_pool_vlan10 ranges=192.168.10.100-192.168.10.200
add name=dhcp_pool_vlan20 ranges=192.168.20.100-192.168.20.200
add name=dhcp_pool_vlan30 ranges=192.168.30.100-192.168.30.200

# Configure DHCP networks
/ip dhcp-server network
add address=192.168.10.0/24 gateway=192.168.10.1 dns-server=8.8.8.8,8.8.4.4
add address=192.168.20.0/24 gateway=192.168.20.1 dns-server=8.8.8.8,8.8.4.4
add address=192.168.30.0/24 gateway=192.168.30.1 dns-server=8.8.8.8,8.8.4.4

# Enable DHCP servers on VLAN interfaces
/ip dhcp-server
add interface=vlan10 address-pool=dhcp_pool_vlan10 name=dhcp_vlan10
add interface=vlan20 address-pool=dhcp_pool_vlan20 name=dhcp_vlan20
add interface=vlan30 address-pool=dhcp_pool_vlan30 name=dhcp_vlan30

# DHCP leases (static assignments for servers)
/ip dhcp-server lease
add address=192.168.40.10 mac-address=AA:BB:CC:DD:EE:FF \
    server=dhcp_vlan40 comment="Web Server"
add address=192.168.40.20 mac-address=11:22:33:44:55:66 \
    server=dhcp_vlan40 comment="Database Server"

Challenges & Solutions

Uneven Load Distribution

Initial PCC load balancing caused uneven distribution due to different ISP speeds. Implemented weighted PCC with connection marking based on bandwidth ratios (2:1 for Fiber:DSL).

Broadcast Storms

Initial VLAN configuration caused broadcast loops. Implemented proper bridge VLAN filtering and RSTP to prevent loops and isolate broadcast domains.

VPN Performance

IPsec VPN had high CPU usage. Migrated to WireGuard for remote access and hardware-accelerated IPsec for site-to-site connections.

QoS for VoIP

Voice quality issues during peak hours. Implemented dedicated VLAN for VoIP with strict priority queuing and bandwidth reservation.
